// Phase-2 design gate harness (scratch; NOT production code).
// Builds a "Space Design Load Summary" Document from a real saved model, reusing the Phase-1
// identity / design-criteria / sizing section builders, plus mock peak sections assembled only from
// the generic SAM.Core.Reporting blocks, and renders it with the production PdfRenderer.
using SAM.Analytical;
using SAM.Analytical.Reporting;
using SAM.Core.Reporting;
using SAM.Core.Reporting.Pdf;
using SAM.Units;
using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;

static class Program
{
    const string TasSource = "SAM.Analytical.Tas";

    // Values NOT persisted today, read once from the fixture's TSD with a read-only reader (probe.cs):
    // BuildingData heating peak for Bathroom_2 = 104.0099 W at 1-based hour 8554 (23 Dec 09:00-10:00).
    sealed class AnnualPeak { public double Load, Db, Res, RH, HR, Inf, Air, Bht, Opq, Glz, OutT, OutRH; public int Hour; }
    static readonly AnnualPeak Bathroom2AnnualHeating = new AnnualPeak { Load = 104.009911, Hour = 8554, Db = 16.0009079, Res = 15.93449, RH = 34.96767, Inf = -93.37398, Air = 0, Bht = -3.15779853, Opq = -7.47790527, Glz = 0, OutT = -2.3, OutRH = 100 };

    static void Main(string[] args)
    {
        string outDir = args.Length > 0 ? args[0] : Directory.GetCurrentDirectory();
        Directory.CreateDirectory(outDir);
        string modelPath = @"C:\TasOut\final1b\open_out.sam";
        AnalyticalModel model = SAM.Core.Convert.ToSAM<AnalyticalModel>(modelPath).First();

        Render(model, "Bathroom_2", UnitStyle.SI, Case.Normal, Path.Combine(outDir, "gate-1-normal-SI.pdf"));
        Render(model, "Bathroom_2", UnitStyle.Imperial, Case.Normal, Path.Combine(outDir, "gate-4-normal-IP.pdf"));
        Render(model, "Studio 1_0", UnitStyle.SI, Case.Legacy, Path.Combine(outDir, "gate-2a-legacy-sentinel-SI.pdf"));
        Render(model, "Corridor_1", UnitStyle.SI, Case.NotSimulated, Path.Combine(outDir, "gate-2b-not-simulated-SI.pdf"));
        Render(model, "Bathroom_2", UnitStyle.SI, Case.Stress, Path.Combine(outDir, "gate-3-long-name-large-values-SI.pdf"));
    }

    enum Case { Normal, Legacy, NotSimulated, Stress }

    static void Render(AnalyticalModel model0, string spaceName, UnitStyle unitStyle, Case @case, string path)
    {
        AnalyticalModel model = new AnalyticalModel(model0, true);
        AdjacencyCluster cluster = model.AdjacencyCluster;
        Space space = cluster.GetSpaces().First(x => x.Name == spaceName);
        double scale = 1.0;
        if (@case == Case.Stress)
        {
            space = new Space(space.Guid, space, "Level 03 — Open-plan Office North-East Wing incl. Breakout / Kitchenette Zone B (Tenant Fit-out Cat A)", space.Location);
            cluster.AddObject(space);
            scale = 118.0;
        }
        if (@case == Case.NotSimulated)
        {
            foreach (SpaceSimulationResult r in cluster.GetResults<SpaceSimulationResult>(space) ?? new List<SpaceSimulationResult>()) cluster.RemoveObject<SpaceSimulationResult>(r.Guid);
        }
        model = new AnalyticalModel(model, cluster);

        DocumentOptions options = new DocumentOptions
        {
            UnitSystem = unitStyle,
            GeneratedAt = new DateTime(2026, 9, 26, 12, 0, 0),
            SoftwareVersion = "design-gate",
            Metadata = new DocumentMetadata { ProjectName = "Part O fixture (final1b/open)" },
        };
        DocumentContext ctx = SAM.Analytical.Reporting.Create.DocumentContext(model, options);
        SpaceDocumentData data = SAM.Analytical.Reporting.Create.SpaceDocumentData(ctx, space);
        IQuantityFormatter f = ctx.Formatter;

        List<SpaceSimulationResult> results = cluster.GetResults<SpaceSimulationResult>(space, TasSource) ?? new List<SpaceSimulationResult>();
        SpaceSimulationResult heat = results.Find(x => x.LoadType() == LoadType.Heating);
        SpaceSimulationResult cool = results.Find(x => x.LoadType() == LoadType.Cooling);

        List<DocumentSection> sections = new List<DocumentSection>
        {
            new SpaceIdentitySectionBuilder().Build(data, ctx),
            ResultSourceSection(f, heat, cool, @case),
            new SpaceDesignCriteriaSectionBuilder().Build(data, ctx),
        };
        if (@case == Case.Stress)
        {
            sections.Insert(1, new DocumentSection("gate-note", "Design gate", new DocumentBlock[] { new NoticeBlock("gate-note", "ILLUSTRATIVE VALUES for layout stress only: Bathroom_2 results scaled ×118 and an invented cooling peak. Not from any result file.", NoticeLevel.Warning) }));
        }
        sections.Add(LoadsSection(f, data, heat, cool, @case, scale));
        if (@case == Case.Normal || @case == Case.Stress)
        {
            sections.Add(ConditionsSection(f, heat, @case, scale));
            sections.Add(BalanceSection(f, heat, @case, scale));
        }

        DocumentMetadata md = new DocumentMetadata(options.Metadata) { Title = "Space Design Load Summary", Subject = space.Name, Date = options.GeneratedAt.Value.Date };
        Document probeDoc = new Document("space-design-load-summary", md, sections, null, options.Style);
        List<string> legend = new List<string>();
        var fvs = probeDoc.FormattedValues().ToList();
        if (fvs.Any(x => x.Availability == Availability.NotAvailable && x.Text == f.NotAvailableText)) legend.Add("— not available");
        if (fvs.Any(x => x.Availability == Availability.NotApplicable && x.Text == f.NotApplicableText)) legend.Add("n/a not applicable");
        string resultLine = heat == null && cool == null
            ? "Results: none in model"
            : string.Format("Results: Tas TSD, converted {0} · model vs results: not recorded", QuantityFormatter.FormatDateTime((heat ?? cool).DateTime));
        DocumentFooter footer = new DocumentFooter(new[]
        {
            string.Format("Model: {0} · SAM {1} · Design loads: from Tas sizing — date/currency not recorded", ctx.Provenance.ModelName ?? "—", ctx.Provenance.SoftwareVersion),
            resultLine,
            string.Format("Generated {0} · Space {1}", QuantityFormatter.FormatDateTime(ctx.Provenance.GeneratedAt), space.Guid.ToString("D")),
        }, legend);
        Document document = new Document("space-design-load-summary", md, sections, footer, options.Style);

        byte[] pdf = new PdfRenderer().Render(document);
        File.WriteAllBytes(path, pdf);
        File.WriteAllText(Path.ChangeExtension(path, ".json"), SAM.Core.Reporting.Convert.ToJson(document));
        Console.WriteLine("{0}  {1:N0} bytes", path, pdf.Length);
    }

    // ---- helpers -------------------------------------------------------------------------------------

    static ReportValue<Quantity> Q(double v, UnitType u, ReportValueSource src = ReportValueSource.TSD)
        => double.IsNaN(v) ? ReportValue<Quantity>.NotAvailable("not in results") : ReportValue<Quantity>.Available(new Quantity(v, u), src, Freshness.Unknown);

    static ReportValue<Quantity> P(SpaceSimulationResult r, SpaceSimulationResultParameter p, UnitType u, double scale = 1.0)
    {
        if (r == null || !r.TryGetValue(p, out double v) || double.IsNaN(v)) return ReportValue<Quantity>.NotAvailable("not in results");
        return Q(v * scale, u);
    }

    static FormattedValue[] Group(IQuantityFormatter f, UnitCategory cat, params ReportValue<Quantity>[] values)
    {
        List<Quantity> qs = new List<Quantity>();
        foreach (var v in values) if (v != null && v.TryGetValue(out Quantity q)) qs.Add(q);
        DisplayUnit du = f.SelectDisplayUnit(cat, qs);
        return values.Select(x => f.Format(x, du)).ToArray();
    }

    static FormattedValue Text(string s) => new FormattedValue(s, null, Availability.Available, Freshness.Unknown, ReportValueSource.TSD);
    static FormattedValue NA(IQuantityFormatter f) => new FormattedValue(f.NotApplicableText, null, Availability.NotApplicable);
    static FormattedValue Missing(IQuantityFormatter f) => new FormattedValue(f.NotAvailableText, null, Availability.NotAvailable);

    static TableRow Row(string label, params FormattedValue[] cells) => new TableRow(new[] { FormattedValue.Label(label) }.Concat(cells));

    // Legacy rule (to be implemented in SAM/SAM_Tas, NOT reporting): a Tas LoadIndex of 0 is outside the
    // verified 1-based range, so every value read at it is Tas's -1 "invalid index" sentinel.
    static bool IsLegacySentinel(SpaceSimulationResult r) => r != null && r.TryGetValue(SpaceSimulationResultParameter.LoadIndex, out int i) && i <= 0;

    static string HourText(int oneBasedHour)
    {
        DateTime start = new DateTime(2018, 1, 1).AddHours(oneBasedHour - 1);
        return string.Format(System.Globalization.CultureInfo.InvariantCulture, "{0:d MMM HH:mm}–{1:HH:mm}", start, start.AddHours(1)).Replace("00:00", oneBasedHour % 24 == 0 ? "24:00" : "00:00");
    }

    static string DesignDayName(SpaceSimulationResult r)
    {
        if (r == null) return null;
        var json = r.ToJsonObject();
        foreach (var ps in json["ParameterSets"]?.AsArray() ?? new System.Text.Json.Nodes.JsonArray())
            foreach (var p in ps?["Parameters"]?.AsArray() ?? new System.Text.Json.Nodes.JsonArray())
                if ((string)p?["Name"] == "Design Day Name") return (string)p["Value"];
        return null;
    }

    // ---- sections ------------------------------------------------------------------------------------

    static DocumentSection ResultSourceSection(IQuantityFormatter f, SpaceSimulationResult heat, SpaceSimulationResult cool, Case @case)
    {
        if (heat == null && cool == null)
        {
            return new DocumentSection("results", "Simulation Results", new DocumentBlock[] { new NoticeBlock("results-missing", "No Tas simulation results in model — simulate to report peaks.", NoticeLevel.Information) }, SectionWidth.Half);
        }
        SpaceSimulationResult any = heat ?? cool;
        string hdd = DesignDayName(heat);
        string cdd = DesignDayName(cool);
        var rows = new List<KeyValueRow>
        {
            new KeyValueRow("Source", Text("Tas (TSD)")),
            new KeyValueRow("Results converted", Text(QuantityFormatter.FormatDateTime(any.DateTime))),
            new KeyValueRow("Model vs results", new FormattedValue("not recorded", null, Availability.NotApplicable), "no provenance record"),
            new KeyValueRow("Heating design day", hdd == null ? Missing(f) : Text(hdd)),
            new KeyValueRow("Cooling design day", cdd == null ? Missing(f) : Text(cdd)),
        };
        return new DocumentSection("results", "Simulation Results", new DocumentBlock[] { new KeyValueBlock("results", null, rows) }, SectionWidth.Half);
    }

    static DocumentSection LoadsSection(IQuantityFormatter f, SpaceDocumentData data, SpaceSimulationResult heat, SpaceSimulationResult cool, Case @case, double scale)
    {
        List<DocumentBlock> blocks = new List<DocumentBlock>();
        ReportValue<Quantity> tbdH = data.Sizing.DesignHeatingLoad, tbdC = data.Sizing.DesignCoolingLoad;
        ReportValue<Quantity> ddH, ddC, anH, anC;
        if (@case == Case.Legacy)
        {
            ddH = ddC = anH = anC = ReportValue<Quantity>.NotAvailable("stored results unusable (legacy −1 sentinel)");
        }
        else if (@case == Case.NotSimulated)
        {
            ddH = ddC = anH = anC = ReportValue<Quantity>.NotAvailable("not simulated");
        }
        else
        {
            ddH = P(heat, SpaceSimulationResultParameter.Load, UnitType.Watt, scale);
            // Cooling: proposed post-PR2A semantics - a zero peak is an explicit 0 (today it is persisted as -1).
            ddC = @case == Case.Stress ? Q(23450 * 1.0, UnitType.Watt) : Q(0, UnitType.Watt);
            anH = Q(Bathroom2AnnualHeating.Load * scale, UnitType.Watt);
            anC = @case == Case.Stress ? Q(19875.5, UnitType.Watt) : Q(0, UnitType.Watt);
        }
        if (@case == Case.Stress)
        {
            tbdH = Q(1139.87451171875 * scale, UnitType.Watt, ReportValueSource.TBD);
            tbdC = Q(24100, UnitType.Watt, ReportValueSource.TBD);
        }
        FormattedValue[] power = Group(f, UnitCategory.Power, tbdH, tbdC, ddH, ddC, anH, anC);
        TableBlock table = new TableBlock("loads", null, new[]
        {
            new TableColumn(string.Empty, alignment: ColumnAlignment.Left),
            new TableColumn("Heating"),
            new TableColumn("Cooling"),
        }, new[]
        {
            Row("Design load (Tas sizing, TBD)", power[0], power[1]),
            Row("Design-day peak (TSD)", power[2], power[3]),
            Row("Annual simulated peak (TSD)", power[4], power[5]),
        });
        blocks.Add(table);
        string note = @case switch
        {
            Case.Legacy => "Stored Tas results for this space predate the corrected conversion (a zero load was written as −1). Re-run the simulation to report peaks.",
            Case.NotSimulated => "Design loads come from Tas sizing. No simulation results are stored for this space.",
            _ => "Design load: zone maximum from Tas sizing (TBD). Design-day and annual peaks: Tas simulation (TSD). Shown side by side; no pass/fail rule is applied.",
        };
        blocks.Add(new NoticeBlock("loads-note", note, @case == Case.Legacy ? NoticeLevel.Warning : NoticeLevel.Note));
        return new DocumentSection("loads", "Heating and Cooling Loads", blocks);
    }

    static DocumentSection ConditionsSection(IQuantityFormatter f, SpaceSimulationResult heat, Case @case, double scale)
    {
        AnnualPeak a = Bathroom2AnnualHeating;
        heat.TryGetValue(SpaceSimulationResultParameter.LoadIndex, out int ddHour);
        FormattedValue[] db = Group(f, UnitCategory.Temperature,
            P(heat, SpaceSimulationResultParameter.DryBulbTempearture, UnitType.Celsius), Q(a.Db, UnitType.Celsius),
            @case == Case.Stress ? Q(24.1, UnitType.Celsius) : ReportValue<Quantity>.NotApplicable("no cooling peak"));
        FormattedValue[] res = Group(f, UnitCategory.Temperature,
            P(heat, SpaceSimulationResultParameter.ResultantTemperature, UnitType.Celsius), Q(a.Res, UnitType.Celsius),
            @case == Case.Stress ? Q(25.3, UnitType.Celsius) : ReportValue<Quantity>.NotApplicable("no cooling peak"));
        FormattedValue[] rh = Group(f, UnitCategory.RelativeHumidity,
            ReportValue<Quantity>.NotAvailable("not stored for heating"), Q(a.RH, UnitType.Percent),
            @case == Case.Stress ? Q(52.0, UnitType.Percent) : ReportValue<Quantity>.NotApplicable("no cooling peak"));
        FormattedValue[] outT = Group(f, UnitCategory.Temperature,
            ReportValue<Quantity>.NotAvailable("design-day outdoor state not stored"), Q(a.OutT, UnitType.Celsius),
            @case == Case.Stress ? Q(28.4, UnitType.Celsius) : ReportValue<Quantity>.NotApplicable("no cooling peak"));
        FormattedValue[] outRH = Group(f, UnitCategory.RelativeHumidity,
            ReportValue<Quantity>.NotAvailable("design-day outdoor state not stored"), Q(a.OutRH, UnitType.Percent),
            @case == Case.Stress ? Q(41.0, UnitType.Percent) : ReportValue<Quantity>.NotApplicable("no cooling peak"));
        TableBlock table = new TableBlock("peak-conditions", null, new[]
        {
            new TableColumn(string.Empty, alignment: ColumnAlignment.Left),
            new TableColumn("Design day", group: "Heating"),
            new TableColumn("Annual", group: "Heating"),
            new TableColumn("Annual", group: "Cooling"),
        }, new[]
        {
            Row("Time", Text(string.Format("hour {0} of design day", (ddHour - 1) % 24 + 1)), Text(HourText(a.Hour)), @case == Case.Stress ? Text("17 Jul 15:00–16:00") : NA(f)),
            Row("Room air temperature", db),
            Row("Room resultant temperature", res),
            Row("Room relative humidity", rh),
            Row("Outdoor dry bulb", outT),
            Row("Outdoor relative humidity", outRH),
        });
        return new DocumentSection("peak-conditions", "Conditions at Peak", new DocumentBlock[]
        {
            table,
            new NoticeBlock("peak-time-note", "Annual times are Tas simulation hours (hour-ending, local standard time, no DST, 365-day year). A design-day peak has no calendar date.", NoticeLevel.Note),
        });
    }

    static DocumentSection BalanceSection(IQuantityFormatter f, SpaceSimulationResult heat, Case @case, double scale)
    {
        AnnualPeak a = Bathroom2AnnualHeating;
        (string label, SpaceSimulationResultParameter p, double annual, double cool)[] rows =
        {
            ("Solar", SpaceSimulationResultParameter.SolarGain, 0, 9120),
            ("Lighting", SpaceSimulationResultParameter.LightingGain, 0, 2210),
            ("Occupancy (sensible)", SpaceSimulationResultParameter.OccupancySensibleGain, 0, 3380),
            ("Equipment (sensible)", SpaceSimulationResultParameter.EquipmentSensibleGain, 0, 4705),
            ("Infiltration / ventilation", SpaceSimulationResultParameter.InfiltrationGain, a.Inf, 610.5),
            ("Air movement (inter-zone)", SpaceSimulationResultParameter.AirMovementGain, a.Air, -120),
            ("Building heat transfer", SpaceSimulationResultParameter.BuildingHeatTransfer, a.Bht, 1540),
            ("External conduction — opaque", SpaceSimulationResultParameter.OpaqueExternalConduction, a.Opq, 1830),
            ("External conduction — glazing", SpaceSimulationResultParameter.GlazingExternalConduction, a.Glz, 1740),
        };
        List<ReportValue<Quantity>> all = new List<ReportValue<Quantity>>();
        List<ReportValue<Quantity>[]> cells = new List<ReportValue<Quantity>[]>();
        double sDD = 0, sAn = 0, sC = 0;
        foreach (var r in rows)
        {
            bool heatingTerm = r.p != SpaceSimulationResultParameter.SolarGain && r.p != SpaceSimulationResultParameter.LightingGain && r.p != SpaceSimulationResultParameter.OccupancySensibleGain && r.p != SpaceSimulationResultParameter.EquipmentSensibleGain;
            ReportValue<Quantity> dd = heatingTerm ? P(heat, r.p, UnitType.Watt, scale) : ReportValue<Quantity>.NotAvailable("not stored for heating");
            ReportValue<Quantity> an = Q(r.annual * scale, UnitType.Watt);
            ReportValue<Quantity> co = @case == Case.Stress ? Q(r.cool, UnitType.Watt) : ReportValue<Quantity>.NotApplicable("no cooling peak");
            if (dd.TryGetValue(out Quantity q1)) sDD += q1.Value; sAn += r.annual * scale; if (@case == Case.Stress) sC += r.cool;
            cells.Add(new[] { dd, an, co }); all.AddRange(new[] { dd, an, co });
        }
        ReportValue<Quantity> netDD = Q(sDD, UnitType.Watt, ReportValueSource.Derived), netAn = Q(sAn, UnitType.Watt, ReportValueSource.Derived);
        ReportValue<Quantity> netC = @case == Case.Stress ? Q(sC, UnitType.Watt, ReportValueSource.Derived) : ReportValue<Quantity>.NotApplicable("no cooling peak");
        all.AddRange(new[] { netDD, netAn, netC });
        List<Quantity> qs = new List<Quantity>();
        foreach (var v in all) if (v.TryGetValue(out Quantity q)) qs.Add(q);
        DisplayUnit du = f.SelectDisplayUnit(UnitCategory.Power, qs);
        List<TableRow> tableRows = rows.Select((r, i) => Row(r.label, cells[i].Select(x => f.Format(x, du)).ToArray())).ToList();
        tableRows.Add(Row("Net (heating: = −load)", f.Format(netDD, du), f.Format(netAn, du), f.Format(netC, du)));
        TableBlock table = new TableBlock("peak-balance", null, new[]
        {
            new TableColumn("Tas zone term", alignment: ColumnAlignment.Left),
            new TableColumn("Design day", du.Symbol, group: "Heating"),
            new TableColumn("Annual", du.Symbol, group: "Heating"),
            new TableColumn("Annual", du.Symbol, group: "Cooling"),
        }, tableRows);
        return new DocumentSection("peak-balance", "Zone Heat Balance at Peak", new DocumentBlock[]
        {
            table,
            new NoticeBlock("balance-note", "Tas zone heat-balance terms at the peak hour: + gain to the room air, − loss. Latent terms are not included.", NoticeLevel.Note),
        });
    }
