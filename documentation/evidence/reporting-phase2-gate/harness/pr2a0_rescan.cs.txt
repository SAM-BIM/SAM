using SAM.Analytical;
using SAM.Analytical.Reporting;
using SAM.Core.Reporting;
using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;

static class Program
{
    static void Main(string[] args)
    {
        string root = args[0];
        string analytical = SAM.Core.Query.Name(typeof(Space).Assembly);
        int files = 0, filesStale = 0, compared = 0, stale = 0, dupAfterLoad = 0, dupAfterResave = 0, failed = 0;
        foreach (string path in Directory.EnumerateFiles(root, "*.sam", SearchOption.AllDirectories).OrderBy(x => x))
        {
            if (path.Contains("website-evidence")) continue;
            AnalyticalModel model;
            try { model = SAM.Core.Convert.ToSAM<AnalyticalModel>(path)?.FirstOrDefault(); } catch { failed++; continue; }
            if (model?.AdjacencyCluster == null) { failed++; continue; }
            files++;
            AdjacencyCluster cluster = model.AdjacencyCluster;
            int fileStale = 0;
            foreach (Space space in cluster.GetSpaces() ?? new List<Space>())
            {
                if (space.GetParameterSets()?.Count(x => x.Name == analytical) > 1) dupAfterLoad++;
                List<SpaceSimulationResult> results = cluster.GetResults<SpaceSimulationResult>(space) ?? new List<SpaceSimulationResult>();
                foreach ((LoadType loadType, SpaceParameter parameter) in new[] { (LoadType.Heating, SpaceParameter.DesignHeatingLoad), (LoadType.Cooling, SpaceParameter.DesignCoolingLoad) })
                {
                    SpaceSimulationResult result = results.Find(x => x.LoadType() == loadType);
                    if (result == null || !result.TryGetValue(SpaceSimulationResultParameter.DesignLoad, out double expected)) continue;
                    compared++;
                    bool has = space.TryGetValue(parameter, out double actual);
                    if (!has || Math.Abs(actual - expected) > 1e-6) { stale++; fileStale++; Console.WriteLine($"  STALE {Path.GetFileName(path)} {space.Name} {loadType}: read {(has ? actual.ToString() : "missing")} vs result {expected}"); }
                }
            }
            // In-memory re-save / reload: no duplicate set may come back (nothing is written to disk).
            AnalyticalModel reloaded = SAM.Core.Create.IJSAMObject<AnalyticalModel>(SAM.Core.Convert.ToString(model));
            foreach (Space space in reloaded.AdjacencyCluster.GetSpaces() ?? new List<Space>())
                if (space.GetParameterSets()?.Count(x => x.Name == analytical) > 1) dupAfterResave++;
            if (fileStale > 0) filesStale++;
        }
        Console.WriteLine($"files {files} (unreadable {failed}); load comparisons {compared}; stale {stale} in {filesStale} files; spaces with >1 '{analytical}' set after load {dupAfterLoad}, after re-save {dupAfterResave}");

        AnalyticalModel open = SAM.Core.Convert.ToSAM<AnalyticalModel>(@"C:\TasOut\final1b\open_out.sam").First();
        Space bathroom = open.AdjacencyCluster.GetSpaces().First(x => x.Name == "Bathroom_2");
        DocumentContext ctx = SAM.Analytical.Reporting.Create.DocumentContext(open, new DocumentOptions { GeneratedAt = new DateTime(2026, 9, 26) });
        SpaceDocumentData data = SAM.Analytical.Reporting.Create.SpaceDocumentData(ctx, bathroom);
        Console.WriteLine($"Bathroom_2: SpaceParameter.DesignHeatingLoad = {bathroom.GetValue<double>(SpaceParameter.DesignHeatingLoad)}; Space Assumptions Sizing.DesignHeatingLoad = {(data.Sizing.DesignHeatingLoad.HasValue ? data.Sizing.DesignHeatingLoad.Value.Value + " W" : "n/a")}, cooling = {(data.Sizing.DesignCoolingLoad.HasValue ? data.Sizing.DesignCoolingLoad.Value.Value + " W" : "n/a")}");
    }
}
