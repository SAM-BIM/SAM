using System; using TSD;
class P {
  static string F(float v){ return v.ToString("R", System.Globalization.CultureInfo.InvariantCulture); }
  static void Dump(string tag, ZoneData z, int h){
    Console.WriteLine("{0} h={1} heat={2} cool={3} db={4} res={5} RH={6} inf={7} bht={8} opq={9} glz={10} sol={11} light={12} occS={13} eqS={14} sens={15} latent={16} vent={17} infil={18} ahu={19}", tag,h,
      F(z.GetHourlyZoneResult(h,5)),F(z.GetHourlyZoneResult(h,6)),F(z.GetHourlyZoneResult(h,1)),F(z.GetHourlyZoneResult(h,3)),F(z.GetHourlyZoneResult(h,17)),
      F(z.GetHourlyZoneResult(h,9)),F(z.GetHourlyZoneResult(h,11)),F(z.GetHourlyZoneResult(h,12)),F(z.GetHourlyZoneResult(h,13)),F(z.GetHourlyZoneResult(h,7)),
      F(z.GetHourlyZoneResult(h,8)),F(z.GetHourlyZoneResult(h,14)),F(z.GetHourlyZoneResult(h,15)),F(z.GetHourlyZoneResult(h,4)),F(z.GetHourlyZoneResult(h,20)),F(z.GetHourlyZoneResult(h,27)),F(z.GetHourlyZoneResult(h,26)),F(z.GetHourlyZoneResult(h,32)));
  }
  static ZoneData Find(Func<int,ZoneData> get, string name, out int idx){ idx=-1; for(int i=1;;i++){ var z=get(i); if(z==null) return null; if(z.name==name){ idx=i; return z; } } }
  static void Peak(string tag, object arr, int col){
    var a=(object[,])arr; Console.WriteLine("{0} peak bounds r[{1}..{2}] c[{3}..{4}] v0={5} v1={6} v2={7}",tag,a.GetLowerBound(0),a.GetUpperBound(0),a.GetLowerBound(1),a.GetUpperBound(1),a[a.GetLowerBound(0),col],a[a.GetLowerBound(0)+1,col],a[a.GetLowerBound(0)+2,col]); }
  static void Main(string[] args){
    var d=new TSDDocument(); var ok=d.openReadOnly(args[0]); Console.WriteLine("open="+ok);
    try {
    var sim=d.SimulationData; string zn=args[1];
    Console.WriteLine("sim firstDay={0} lastDay={1}", sim.firstDay, sim.lastDay);
    var bd=sim.GetBuildingData(); int bi; var bz=Find(i=>bd.GetZoneData(i),zn,out bi);
    Console.WriteLine("bd zone idx="+bi+" guid="+(bz==null?"":bz.zoneGUID));
    for(int k=0;k<4;k++){ var h=sim.GetHeatingDesignData(k); if(h==null)break; Console.WriteLine("HDD["+k+"] "+h.name); }
    for(int k=0;k<4;k++){ var c=sim.GetCoolingDesignData(k); if(c==null)break; Console.WriteLine("CDD["+k+"] "+c.name); }
    var hdd=sim.GetHeatingDesignData(0); int hi; var hz=Find(i=>hdd.GetZoneData(i),zn,out hi);
    Peak("HDD heat", hdd.GetPeakZoneGains(new short[]{5}), hi-1);
    Peak("BD heat", bd.GetPeakZoneGains(new short[]{5}), bi-1);
    Peak("BD cool", bd.GetPeakZoneGains(new short[]{6}), bi-1);
    var cdd=sim.GetCoolingDesignData(0); int ci; var cz=Find(i=>cdd.GetZoneData(i),zn,out ci);
    Peak("CDD cool", cdd.GetPeakZoneGains(new short[]{6}), ci-1);
    foreach(var s in args[2].Split(',')){ int h=int.Parse(s); Dump("HDD",hz,h); }
    foreach(var s in args[3].Split(',')){ int h=int.Parse(s); Dump("BD ",bz,h); Console.WriteLine("   ext T={0} RH={1}", F(bd.GetHourlyBuildingResult(h,2)),F(bd.GetHourlyBuildingResult(h,1))); }
    } finally { d.close(); }
  }
}
